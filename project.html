🏆 Find Maximum

Step through the Find Max program. Screenshot AL and flags after each loop iteration (especially last).

.model small
.data
    arr   db 12,7,25,9,18,3
    count db 6
.code
main:
    mov ax,@data
    mov ds,ax
    mov cl,count
    dec cl
    mov si,offset arr
    mov al,[si]
    inc si
search:
    cmp al,[si]
    jae not_bigger
    mov al,[si]
not_bigger:
    inc si
    dec cl
    jnz search
    mov ah,4ch
    int 21h
end main

Write a complete MASM program that adds three DWORD values and stores the result.

.model flat, stdcall
.stack 4096
ExitProcess PROTO, dwExitCode:DWORD

.data
    a   DWORD 5
    b   DWORD 10
    c   DWORD 3
    sum DWORD ?

.code
main PROC
    mov eax, a
    add eax, b
    add eax, c
    mov sum, eax
    INVOKE ExitProcess, 0
main ENDP
END main
